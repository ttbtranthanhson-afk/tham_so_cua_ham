def prime(n):
    C = 0
    k = 1
    while k < n:
        if n % k == 0:
            C = C + 1
        k = k + 1
    if C == 1:
        return True
    else:
        return False

# Chương trình chính
n = int(input("Nhập số tự nhiên n: "))
for k in range(1, n+1):
    if prime(k):
        print(k, end=" ")

def tongduong(A):
    S = 0
    for k in A:
        if k > 0:
            S = S + k
    return S

# Chương trình chính
